module load gatk

SPECIES="-" 
SAMPLE=$(sed -n "${SLURM_ARRAY_TASK_ID}p" utils/sample_list${SPECIES}.txt)
REFERENCE="$utils/${SPECIES}/reference"
BAM_DIR="$dedup/lampyris/"
OUT_DIR="$gatk/${SPECIES}"

mkdir -p "${OUTDIR}"

  -R ${REFERENCE}/*.fasta \
  -I dedup/lampyris/${SAMPLE}_dedup_q20.bam \
  -O gatk/lampyris/gvcf/${SAMPLE}.g.vcf.gz \
  -ERC GVCF
