#!/bin/bash
#SBATCH --job-name=dedup
#SBATCH --partition=krypton
#SBATCH --ntasks=1
#SBATCH --mem=32G
#SBATCH --array=1-67
#SBATCH -o logs/dedup_%A_%a.out
#SBATCH -e err/dedup_%A_%a.err

module load gnu python python-packages
module load picard samtools

SPECIES="-" #insert either lampyris / lamprohiza

SAMPLE=$(sed -n "${SLURM_ARRAY_TASK_ID}p" utils/sample_list${SPECIES}.txt)
ALIGNMENTS="$02_alignments/${SPECIES}/"
OUTDIR="$03_dedup/${SPECIES}"

mkdir -p {OUTDIR}

picard MarkDuplicates \
  I=${ALIGNMENTS}/${SAMPLE}_sorted.bam \
  O={OUTDIR}/${SAMPLE}_dedup.bam \
  M={OUTDIR}/${SAMPLE}_metrics.txt \
  REMOVE_DUPLICATES=true \
  VALIDATION_STRINGENCY=SILENT

samtools view -b -q 20 dedup/${SPECIES}/${SAMPLE}_dedup.bam \
  > dedup/${SPECIES}/${SAMPLE}_dedup_q20.bam

samtools index dedup/${SPECIES}/${SAMPLE}_dedup_q20.bam
