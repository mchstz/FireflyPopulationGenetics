#!/bin/bash
#SBATCH --job-name=bwa
#SBATCH --partition=krypton
#SBATCH --ntasks=1
#SBATCH --cpus-per-task=8
#SBATCH --mem=16G
#SBATCH --array=1-67
#SBATCH -o 01_preprocessinglogs/bwa_%A_%a.out
#SBATCH -e 01_preprocessing/err/bwa_%A_%a.err

module load gnu python python-packages
module load bwa samtools

SAMPLE=$(sed -n "${SLURM_ARRAY_TASK_ID}p" scripts/sample_list.txt)
TRIM_DIR="$01_preprocessign/01_trimmed_reads/"
REFERENCE="$01_preprocessing/utils/
OUT_DIR="$01_preprocessign/02_alignments/"

mkdir -p ${OUTDIR}

RG="@RG\tID:${SAMPLE}\tSM:${SAMPLE}\tPL:ILLUMINA"

bwa mem -t 8 \
  -R "${RG}" \
  ${REFERENCE} \
  ${TRIM_DIR}/${SAMPLE}_1_val_1.fq.gz \
  ${TRIM_DIR}/${SAMPLE}_2_val_2.fq.gz | \
samtools view -Sb - > ${OUT_DIR}/${SAMPLE}.bam

samtools sort -o ${OUT_DIR}/${SAMPLE}_sorted.bam ${OUT_DIR}/${SAMPLE}.bam
samtools index ${OUT_DIR}${SAMPLE}_sorted.bam
rm ${OUT_DIR}/${SAMPLE}.bam
